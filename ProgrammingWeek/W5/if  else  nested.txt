#if
def is_x_positive(x):
    if x > 0:
        print("x is positive")
    if x < 0:
        print("x is negative")
    if x == 0:
        print("x is zero")

# Nested
def is_x_positive_nested(x):
    if x > 0:
        print("x is positive")
    else:
        if x < 0:
            print("x is negative")
        else:
            print("x is zero")
