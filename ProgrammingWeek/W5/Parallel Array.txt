# เริ่มจาก list ว่าง แล้ว append
def setup_random_points(n):
    xs, ys, s = [], [], []         # s = speed
    i = 0
    while i < n:
        xs.append(random(100, 900))
        ys.append(random(100, 500))
        s.append(random(1, 5))
        i = i + 1
    return xs, ys, s


def draw_with_speed(xs, ys, s, n):
    i = 0
    while i < n:
        draw_circle(xs[i], ys[i], 20)
        ys[i] = ys[i] + s[i]
        i = i + 1
